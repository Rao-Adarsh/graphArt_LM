You are given Python source that defines a PyTorch nn.Module and creates one instance of it.
Without running the code, list the operations its forward pass performs, as torch.fx
would trace them.

Do NOT work out how the operations connect. Only list them, in order.

Rules
- One line per traced operation. The op is one of:
    placeholder     an input to forward()
    get_attr        a stored parameter or buffer read directly
    call_module     a layer, e.g. nn.Linear or nn.Conv2d
    call_function   a free function, e.g. add, cat, relu, gelu
    call_method     a tensor method, e.g. view, mean, flatten
    output          the value returned
- A line is:
    <id> <op> <target> [<Class>(<params>)] [<shape>]
  * <target> is the attribute path for call_module (fc1, blocks.0, layer1.0.conv1),
    and the normalised function name otherwise (add, cat, relu, gelu, view).
  * <Class>(<params>) appears for call_module only. Write exactly these parameters,
    named this way, and no others; a layer with no parameters is the bare class name:
      Linear(in=8,out=16,bias=True)        Conv2d(in=3,out=16,k=3,s=1,p=1)
      BatchNorm2d(features=16)             LayerNorm(shape=[32])
      MaxPool2d(k=2,s=2)                   AdaptiveAvgPool2d(out=1)
      Flatten(start=1,end=-1)              Dropout(p=0.1)
      Embedding(num=100,dim=16)            ReLU   GELU   SiLU   Sigmoid   Tanh
    Every Conv/ConvTranspose takes in,out,k,s,p (and g only when groups is not 1),
    never bias. BatchNorm and GroupNorm take features. MaxPool/AvgPool take k,s.
    AdaptiveAvgPool/AdaptiveMaxPool take out. A layer not listed here, such as
    MultiheadAttention or Upsample, is the bare class name. A list value is bracketed
    and joined with x: k=[3x5].
  * <shape> is that node's output shape in square brackets, e.g. [2,16].
    The shape going in is given by example_input at the bottom of the source.
- Function names are normalised: `x + y`, torch.add(x, y) and x.add(y) are all `add`;
  F.relu, torch.relu and x.relu() are all `relu`. An activation written as a layer
  (nn.ReLU) stays a call_module node with its attribute path.
- Ids start at 0, in the order the operations happen.
- Unroll every loop. Expand nn.Sequential and nn.ModuleList into one line per layer,
  with paths like blocks.0, blocks.1. Omit any branch that cannot run, such as an
  `if` on an attribute that is False.

import torch
from torch import nn
from torch.nn import functional as F


class BasicConv(nn.Module):
    def __init__(self, in_channels, out_channels, **kwargs):
        super().__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, bias=False, **kwargs)
        self.bn = nn.BatchNorm2d(out_channels, eps=0.001)
        self.act = nn.Hardswish()

    def forward(self, x):
        x = self.conv(x)
        x = self.bn(x)
        return self.act(x)


class Inception(nn.Module):
    def __init__(self, in_channels, ch1, ch2, pool_features):
        super().__init__()
        self.branch1 = BasicConv(in_channels, ch1, kernel_size=1)
        self.branch2_1 = BasicConv(in_channels, ch2, kernel_size=1)
        self.branch2_2 = BasicConv(ch2, ch2, kernel_size=5, padding=2)
        self.pool = nn.AvgPool2d(kernel_size=3, stride=1, padding=1)
        self.branch_pool = BasicConv(in_channels, pool_features, kernel_size=1)

    def _forward(self, x):
        b1 = self.branch1(x)
        y2 = self.branch2_1(x)
        y2 = self.branch2_2(y2)
        bp = self.pool(x)
        bp = self.branch_pool(bp)
        return [b1, y2, bp]

    def forward(self, x):
        outputs = self._forward(x)
        return torch.cat(outputs, 1)


net = Inception(32, 24, 24, 16)
example_input = torch.randn(1, 32, 8, 8)

import torch
from torch import nn
from torch.nn import functional as F


class BasicConv(nn.Module):
    def __init__(self, in_channels, out_channels, **kwargs):
        super().__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, bias=False, **kwargs)
        self.bn = nn.BatchNorm2d(out_channels, eps=0.001)
        self.act = nn.ReLU()

    def forward(self, feats):
        feats = self.conv(feats)
        feats = self.bn(feats)
        return self.act(feats)


class SqueezeExcitation(nn.Module):
    def __init__(self, channels, squeeze_channels):
        super().__init__()
        self.avgpool = nn.AdaptiveAvgPool2d(1)
        self.fc1 = nn.Conv2d(channels, squeeze_channels, 1)
        self.fc2 = nn.Conv2d(squeeze_channels, channels, 1)
        self.activation = nn.ReLU()
        self.scale_activation = nn.Hardsigmoid()

    def forward(self, feats):
        scale = self.avgpool(feats)
        scale = self.fc1(scale)
        scale = self.activation(scale)
        scale = self.fc2(scale)
        scale = self.scale_activation(scale)
        return scale * feats


class MobileBlock(nn.Module):
    def __init__(self, inp, oup, stride, expand_ratio, use_se=False):
        super().__init__()
        self.stride = stride

        hidden_dim = int(round(inp * expand_ratio))
        self.residual = self.stride == 1 and inp == oup

        layers = []
        if expand_ratio != 1:
            layers.append(BasicConv(inp, hidden_dim, kernel_size=1))
        layers.append(BasicConv(hidden_dim, hidden_dim, kernel_size=3, stride=stride, padding=1, groups=hidden_dim))
        if use_se:
            layers.append(SqueezeExcitation(hidden_dim, max(1, hidden_dim // 4)))
        layers.append(nn.Conv2d(hidden_dim, oup, 1, 1, 0, bias=False))
        layers.append(nn.BatchNorm2d(oup))
        self.conv = nn.Sequential(*layers)

    def forward(self, feats):
        if self.residual:
            return feats + self.conv(feats)
        return self.conv(feats)


block = MobileBlock(32, 32, 1, 4, use_se=True)
example_input = torch.randn(1, 32, 28, 28)
