You are given Python source that defines a PyTorch nn.Module, and the list of operations
its forward pass performs. Without running the code, work out how those operations
connect.

The nodes are given to you and are correct. Do NOT repeat them, change them or add to
them. Produce only the edges.

Rules
- An edge A -> B means node A's output is an argument of node B, and names the argument
  slots it fills:
    3 -> 5 : arg0
  If one value fills several slots of the same operation, that is ONE edge listing all
  of them:
    4 -> 5 : arg0 arg1
  So `x + x`, where both operands are the same value, is a single edge with two slots.
- Slots count tensor arguments only, left to right. In cat([a, b], dim=1) the tensors
  a and b are arg0 and arg1; dim is not a slot.
- Every node except the inputs must receive its operands from somewhere, and every node
  must lead towards the output.
- Work from the source: follow which variable holds which node's result as forward()
  proceeds, and read the connections off that.

HOW TO WORK IT OUT

Simulate forward() one statement at a time, as if you were the interpreter, keeping a
record of which node each variable holds. When a statement calls a helper -- a module
or function of this source, or an nn.Sequential of them -- step into it: its lines run
with its own variable names, its parameters holding what it was called with. Every
given node is made by exactly one line of this simulation.

Write the simulation under a heading SIMULATION, one line per node:
    <statement>   ->   n<id> <op> <target>   <- <variable>=n<id> arg<k>, ...
(an input no variable holds -- a nested call, the previous layer of an nn.Sequential --
is just n<id> arg<k>). Then write the edges under the heading EDGES, in exactly this
format, one edge per line, and nothing after them:

EDGES
0 -> 1 : arg0
1 -> 2 : arg0

import torch
from torch import nn
from torch.nn import functional as F


class BasicConv(nn.Module):
    def __init__(self, in_channels, out_channels, **kwargs):
        super().__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, bias=False, **kwargs)
        self.bn = nn.BatchNorm2d(out_channels, eps=0.001)
        self.act = nn.Hardswish()

    def forward(self, x):
        x = self.conv(x)
        x = self.bn(x)
        return self.act(x)


class Inception(nn.Module):
    def __init__(self, in_channels, ch1, ch2, pool_features):
        super().__init__()
        self.branch1 = BasicConv(in_channels, ch1, kernel_size=1)
        self.branch2_1 = BasicConv(in_channels, ch2, kernel_size=1)
        self.branch2_2 = BasicConv(ch2, ch2, kernel_size=5, padding=2)
        self.pool = nn.AvgPool2d(kernel_size=3, stride=1, padding=1)
        self.branch_pool = BasicConv(in_channels, pool_features, kernel_size=1)

    def _forward(self, x):
        b1 = self.branch1(x)
        y2 = self.branch2_1(x)
        y2 = self.branch2_2(y2)
        bp = self.pool(x)
        bp = self.branch_pool(bp)
        return [b1, y2, bp]

    def forward(self, x):
        outputs = self._forward(x)
        return torch.cat(outputs, 1)


net = Inception(32, 24, 24, 16)
example_input = torch.randn(1, 32, 8, 8)

import torch
from torch import nn
from torch.nn import functional as F


class BasicConv(nn.Module):
    def __init__(self, in_channels, out_channels, **kwargs):
        super().__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, bias=False, **kwargs)
        self.bn = nn.BatchNorm2d(out_channels, eps=0.001)
        self.act = nn.ReLU()

    def forward(self, feats):
        feats = self.conv(feats)
        feats = self.bn(feats)
        return self.act(feats)


class SqueezeExcitation(nn.Module):
    def __init__(self, channels, squeeze_channels):
        super().__init__()
        self.avgpool = nn.AdaptiveAvgPool2d(1)
        self.fc1 = nn.Conv2d(channels, squeeze_channels, 1)
        self.fc2 = nn.Conv2d(squeeze_channels, channels, 1)
        self.activation = nn.ReLU()
        self.scale_activation = nn.Hardsigmoid()

    def forward(self, feats):
        scale = self.avgpool(feats)
        scale = self.fc1(scale)
        scale = self.activation(scale)
        scale = self.fc2(scale)
        scale = self.scale_activation(scale)
        return scale * feats


class MobileBlock(nn.Module):
    def __init__(self, inp, oup, stride, expand_ratio, use_se=False):
        super().__init__()
        self.stride = stride

        hidden_dim = int(round(inp * expand_ratio))
        self.residual = self.stride == 1 and inp == oup

        layers = []
        if expand_ratio != 1:
            layers.append(BasicConv(inp, hidden_dim, kernel_size=1))
        layers.append(BasicConv(hidden_dim, hidden_dim, kernel_size=3, stride=stride, padding=1, groups=hidden_dim))
        if use_se:
            layers.append(SqueezeExcitation(hidden_dim, max(1, hidden_dim // 4)))
        layers.append(nn.Conv2d(hidden_dim, oup, 1, 1, 0, bias=False))
        layers.append(nn.BatchNorm2d(oup))
        self.conv = nn.Sequential(*layers)

    def forward(self, feats):
        if self.residual:
            return feats + self.conv(feats)
        return self.conv(feats)


block = MobileBlock(32, 32, 1, 4, use_se=True)
example_input = torch.randn(1, 32, 28, 28)
